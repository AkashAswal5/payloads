## ========================================
## SQLALCHEMY (PYTHON)
## ========================================

# Query manipulation

# ORM method bypass

## ========================================
## ELOQUENT (PHP/LARAVEL)
## ========================================

# Mass assignment
User::create($request->all());  // Can set any attribute

# Query manipulation
User::where('id', $request->input('id'))->get();
User::whereRaw("id = " . $id)->get();  # Vulnerable

# SQL injection in raw queries
DB::select("SELECT * FROM users WHERE id = " . $id);

## ========================================
## ACTIVE RECORD (RUBY ON RAILS)
## ========================================

# Mass assignment
User.create(params[:user])  # Can set protected attributes

# SQL injection
User.where("username = '#{params[:username]}'")  # Vulnerable

# Safe parameterization
User.where("username = ?", params[:username])

## ========================================
## MONGODB/MONGOOSE
## ========================================

# NoSQL injection
{username: 'admin', password: {$ne: null}}
{username: {$regex: '.*'}}
{$where: 'this.username == "admin"'}

# Mongoose query injection
User.find({username: req.body.username})
// If req.body.username = {$ne: null}, returns all users

## ========================================
## TYPE JUGGLING IN ORM
## ========================================

# PHP/Laravel type juggling
// Input: id[]=1

# JavaScript type coercion
// Input: id[$ne]=null
