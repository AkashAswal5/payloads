## ========================================
## AUTOMATION SCRIPT
## ========================================

#!/bin/bash
TARGET="target.com"
IP="1.2.3.4"

while read vhost; do
  echo "Testing: $vhost"
  response=$(curl -s -o /dev/null -w "%{http_code}" -H "Host: $vhost.$TARGET" http://$IP/)
  if [ "$response" != "404" ]; then
    echo "[+] Found: $vhost.$TARGET (HTTP $response)"
  fi
done < vhosts.txt
