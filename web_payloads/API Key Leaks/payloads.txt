## ========================================
## COMMON MISTAKES
## ========================================

# Hardcoded in source
const API_KEY = "sk_live_xxxx";
api_key = "AKIA1234567890ABCDEF"
$apiSecret = "abcdef123456";

# In comments
// API Key: sk_test_12345
# Secret: ghp_xxxxxxxxxxxx

# In URLs
?api_key=AKIA1234567890ABCDEF
?access_token=sk_live_xxxx

## ========================================
## TESTING METHODOLOGY
## ========================================

# 1. Search source code
grep -r "api_key" .
grep -r "AKIA" .
grep -r "sk_live" .

# 2. Check configuration files
cat .env
cat config.json
cat secrets.yml

# 3. Check version control
git log -p | grep -i "api_key"
git log -p | grep -i "password"

# 4. Check commit history
git grep "api_key" $(git rev-list --all)

# 5. Use tools
trufflehog filesystem .
gitleaks detect --source .
