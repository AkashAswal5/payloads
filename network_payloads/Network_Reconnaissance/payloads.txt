# Network Reconnaissance Payloads

## NMAP - Network Scanning

### Basic Scanning
nmap 192.168.1.0/24                                    # Simple ping scan
nmap -sn 192.168.1.0/24                               # Ping sweep (no port scan)
nmap -sL 192.168.1.0/24                               # List targets without scanning
nmap -Pn 192.168.1.100                                # Skip host discovery

### Port Scanning Techniques
nmap -sS 192.168.1.100                                # TCP SYN scan (stealth)
nmap -sT 192.168.1.100                                # TCP Connect scan
nmap -sU 192.168.1.100                                # UDP scan
nmap -sA 192.168.1.100                                # TCP ACK scan (firewall detection)
nmap -sW 192.168.1.100                                # TCP Window scan
nmap -sM 192.168.1.100                                # TCP Maimon scan
nmap -sN 192.168.1.100                                # TCP Null scan
nmap -sF 192.168.1.100                                # TCP FIN scan
nmap -sX 192.168.1.100                                # TCP Xmas scan

### Port Specification
nmap -p 80 192.168.1.100                              # Scan single port
nmap -p 80,443,8080 192.168.1.100                     # Scan specific ports
nmap -p 1-65535 192.168.1.100                         # Scan all ports
nmap -p- 192.168.1.100                                # Scan all 65535 ports
nmap -p U:53,T:21-25,80 192.168.1.100                 # Mix TCP and UDP
nmap --top-ports 100 192.168.1.100                    # Scan top 100 ports
nmap -F 192.168.1.100                                 # Fast scan (100 common ports)

### Service and Version Detection
nmap -sV 192.168.1.100                                # Service version detection
nmap -sV --version-intensity 9 192.168.1.100          # Aggressive version detection
nmap -sV --version-light 192.168.1.100                # Light version detection
nmap -sV --version-all 192.168.1.100                  # Try all probes

### OS Detection
nmap -O 192.168.1.100                                 # OS detection
nmap -O --osscan-guess 192.168.1.100                  # Aggressive OS guessing
nmap -A 192.168.1.100                                 # Aggressive scan (OS, version, scripts, traceroute)

### Timing and Performance
nmap -T0 192.168.1.100                                # Paranoid (IDS evasion)
nmap -T1 192.168.1.100                                # Sneaky
nmap -T2 192.168.1.100                                # Polite
nmap -T3 192.168.1.100                                # Normal (default)
nmap -T4 192.168.1.100                                # Aggressive
nmap -T5 192.168.1.100                                # Insane
nmap --min-rate 100 192.168.1.100                     # Min packets per second
nmap --max-rate 1000 192.168.1.100                    # Max packets per second

### NSE Scripts (Nmap Scripting Engine)
nmap --script=default 192.168.1.100                   # Default scripts
nmap --script=vuln 192.168.1.100                      # Vulnerability scripts
nmap --script=exploit 192.168.1.100                   # Exploit scripts
nmap --script=auth 192.168.1.100                      # Authentication scripts
nmap --script=brute 192.168.1.100                     # Brute force scripts
nmap --script=discovery 192.168.1.100                 # Discovery scripts
nmap --script=malware 192.168.1.100                   # Malware detection
nmap --script=safe 192.168.1.100                      # Safe scripts only
nmap --script="http-*" 192.168.1.100                  # All HTTP scripts
nmap --script-args http.useragent="Mozilla" 192.168.1.100  # Script arguments

### Firewall/IDS Evasion
nmap -f 192.168.1.100                                 # Fragment packets
nmap -D RND:10 192.168.1.100                          # Decoy scan (10 random)
nmap -D 192.168.1.5,192.168.1.6,ME 192.168.1.100      # Specific decoys
nmap -S 192.168.1.50 192.168.1.100                    # Spoof source address
nmap -g 53 192.168.1.100                              # Use source port 53
nmap --source-port 53 192.168.1.100                   # Source port specification
nmap --data-length 25 192.168.1.100                   # Append random data
nmap --mtu 16 192.168.1.100                           # Set MTU
nmap --badsum 192.168.1.100                           # Send packets with bad checksums
nmap --randomize-hosts 192.168.1.0/24                 # Randomize target order

### Output Options
nmap -oN scan.txt 192.168.1.100                       # Normal output
nmap -oX scan.xml 192.168.1.100                       # XML output
nmap -oG scan.gnmap 192.168.1.100                     # Grepable output
nmap -oA scan 192.168.1.100                           # All formats
nmap -v 192.168.1.100                                 # Verbose
nmap -vv 192.168.1.100                                # Very verbose
nmap -d 192.168.1.100                                 # Debugging
nmap --reason 192.168.1.100                           # Show reason for port state

### Comprehensive Scans
nmap -sS -sV -O -p- --script=vuln -T4 192.168.1.100                    # Full comprehensive
nmap -sS -sU -sV -O -p- --script=default,vuln -T4 192.168.1.100        # TCP + UDP comprehensive
nmap -sS -sV -O -p- --script="default,vuln,exploit" -oA full_scan 192.168.1.100  # Full pentest scan

### Masscan - Fast Port Scanner
masscan 192.168.1.0/24 -p0-65535 --rate 10000                          # Ultra fast scan
masscan 192.168.1.0/24 -p80,443,8080 --rate 1000                       # Fast specific ports
masscan 0.0.0.0/0 -p80,443 --rate 100000 --excludefile exclude.txt    # Internet-wide scan
masscan 192.168.1.0/24 -p0-65535 --banners --rate 10000                # With banner grabbing

### Zmap - Internet Scanner
zmap -p 80 192.168.1.0/24                                              # Single port scan
zmap -p 443 -o results.csv 192.168.1.0/24                              # Output to CSV
zmap -p 80 -B 10M 192.168.1.0/24                                       # 10 Mbps bandwidth
zmap -p 22 --probe-module=ssh 192.168.1.0/24                           # SSH probing

### Unicornscan - Advanced Scanner
unicornscan -mT 192.168.1.100:1-65535                                  # TCP scan
unicornscan -mU 192.168.1.100:1-1024                                   # UDP scan
unicornscan -mTs 192.168.1.100:a                                       # TCP SYN all ports
unicornscan -r 10000 -mT 192.168.1.100:1-65535                         # Fast rate

### Hping3 - Packet Crafting
hping3 -S 192.168.1.100 -p 80                                          # SYN scan port 80
hping3 -A 192.168.1.100 -p 80                                          # ACK scan
hping3 -F -P -U 192.168.1.100 -p 80                                    # FIN, PUSH, URG scan
hping3 -1 192.168.1.100                                                # ICMP ping
hping3 -2 192.168.1.100 -p 53                                          # UDP scan
hping3 -S --flood 192.168.1.100 -p 80                                  # SYN flood (DoS)
hping3 -c 1000000 -d 120 -S -w 64 -p 80 --flood --rand-source 192.168.1.100  # Spoofed SYN flood

### Banner Grabbing
nc -v 192.168.1.100 80                                                 # Netcat banner grab
echo "" | nc -v -n -w1 192.168.1.100 21-25                             # Port range
telnet 192.168.1.100 80                                                # Telnet banner
curl -I http://192.168.1.100                                           # HTTP headers
curl -v telnet://192.168.1.100:21                                      # FTP banner via curl

### DNS Reconnaissance
nslookup target.com                                                    # Basic DNS lookup
nslookup -type=mx target.com                                           # Mail servers
nslookup -type=ns target.com                                           # Name servers
nslookup -type=any target.com                                          # All records
dig target.com                                                         # DNS query
dig target.com ANY                                                     # All records
dig target.com MX                                                      # Mail exchange
dig target.com NS                                                      # Name servers
dig target.com TXT                                                     # TXT records
dig @8.8.8.8 target.com                                                # Query specific DNS
dig +short target.com                                                  # Short output
dig -x 192.168.1.100                                                   # Reverse DNS
host target.com                                                        # Simple DNS lookup
host -t mx target.com                                                  # MX records
host -t ns target.com                                                  # NS records
host -a target.com                                                     # All records

### DNS Zone Transfer
dig @ns1.target.com target.com axfr                                    # Zone transfer
host -l target.com ns1.target.com                                      # Zone transfer (host)
nmap --script dns-zone-transfer --script-args dns-zone-transfer.domain=target.com -p 53 ns1.target.com
fierce --domain target.com                                             # DNS enumeration
dnsrecon -d target.com -t axfr                                         # Zone transfer attempt
dnsrecon -d target.com -t std                                          # Standard enumeration
dnsenum target.com                                                     # Comprehensive DNS enum

### Subdomain Discovery
sublist3r -d target.com                                                # Sublist3r
amass enum -d target.com                                               # OWASP Amass
amass enum -passive -d target.com                                      # Passive only
amass enum -active -d target.com -brute                                # Active + brute
subfinder -d target.com                                                # Subfinder
assetfinder --subs-only target.com                                     # Assetfinder
findomain -t target.com                                                # Findomain
gobuster dns -d target.com -w wordlist.txt                             # Gobuster DNS
gobuster dns -d target.com -w wordlist.txt -t 50                       # 50 threads

### Network Traffic Analysis
tcpdump -i eth0                                                        # Capture all traffic
tcpdump -i eth0 -n                                                     # No DNS resolution
tcpdump -i eth0 -nn                                                    # No DNS/port resolution
tcpdump -i eth0 -w capture.pcap                                        # Write to file
tcpdump -r capture.pcap                                                # Read from file
tcpdump -i eth0 port 80                                                # Filter by port
tcpdump -i eth0 host 192.168.1.100                                     # Filter by host
tcpdump -i eth0 src 192.168.1.100                                      # Source filter
tcpdump -i eth0 dst 192.168.1.100                                      # Destination filter
tcpdump -i eth0 'tcp[tcpflags] & (tcp-syn) != 0'                       # SYN packets only
tshark -i eth0                                                         # Wireshark CLI
tshark -i eth0 -w capture.pcap                                         # Write capture
tshark -r capture.pcap -Y "http"                                       # HTTP filter
tshark -r capture.pcap -Y "dns"                                        # DNS filter
