# Network Evasion and Firewall/IDS Bypass Payloads

## Firewall Detection

### ACK Scan (Firewall Detection)
nmap -sA 192.168.1.100                                                 # ACK scan
nmap -sA -p 1-65535 192.168.1.100                                      # Full port ACK scan
hping3 -A -p 80 192.168.1.100                                          # ACK probe

### FIN Scan (Firewall Evasion)
nmap -sF 192.168.1.100                                                 # FIN scan
hping3 -F -p 80 192.168.1.100                                          # FIN probe

### NULL Scan
nmap -sN 192.168.1.100                                                 # NULL scan
hping3 -p 80 192.168.1.100                                             # No flags set

### Xmas Scan
nmap -sX 192.168.1.100                                                 # Xmas scan
hping3 -F -P -U -p 80 192.168.1.100                                    # FIN, PUSH, URG flags

### Idle Scan (Zombie Scan)
nmap -sI zombie.host.com:80 192.168.1.100                              # Idle scan
nmap -Pn -sI 192.168.1.50 192.168.1.100                                # With zombie IP

## Packet Fragmentation

### IP Fragmentation
nmap -f 192.168.1.100                                                  # Fragment packets (8 bytes)
nmap -f -f 192.168.1.100                                               # 16-byte fragments
nmap --mtu 16 192.168.1.100                                            # Custom MTU
nmap --mtu 24 192.168.1.100                                            # 24-byte MTU
hping3 -f -p 80 192.168.1.100                                          # Fragmented packets

### Fragroute
fragroute 192.168.1.100                                                # Intercept and fragment
# Edit /etc/fragroute.conf
# ip_frag 24
# tcp_seg 24

### Fragment Overlap Attack
# Using Scapy
python3 << 'EOF'
from scapy.all import *
packet1 = IP(dst="192.168.1.100", flags="MF", frag=0)/TCP(dport=80)/"AAAA"
packet2 = IP(dst="192.168.1.100", frag=1)/("BBBB")
send([packet1, packet2])
EOF
